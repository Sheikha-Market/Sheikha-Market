## 💡 Usage Examples

### **Example 1: Automatic Hardware Optimization**
```javascript
const UniversalHAL = require('./js/sheikha-universal-hal');
const hal = new UniversalHAL();

// Get hardware profile
const report = hal.generateReport();
console.log(`Architecture: ${report.systemInfo.architectureInfo.name}`);

// Get optimizations for your hardware
const optimizations = hal.optimizeHardwareResources();
```

### **Example 2: Run Evolution Cycle**
```javascript
const ContinuousEvolution = require('./js/sheikha-continuous-evolution');
const engine = new ContinuousEvolution();

// Run 5 evolution cycles with continuous improvement
await engine.runContinuousEvolution(5);
```

### **Example 3: Intelligent Routing**
```javascript
const Routing = require('./js/sheikha-routing-carrier-system');
const router = new Routing();

// Get optimal route for your needs
const route = router.selectOptimalRoute('performance');
console.log(`Primary: ${route.primary.name}`);
console.log(`Backup: ${route.backup.name}`);

// Route a packet
const result = router.routePacket({
  id: 'PKT-001',
  source: '192.168.1.1',
  destination: '10.0.0.1'
}, 'balanced');
```

### **Example 4: Complete System Orchestration**
```javascript
const Orchestrator = require('./js/sheikha-universal-os-orchestrator');
const os = new Orchestrator();

// Boot complete system
os.boot();

// Get system status
const status = os.getSystemStatus();

// Run evolution cycle
await os.runEvolutionCycle();

// Display dashboard
os.displayDashboard();
```
